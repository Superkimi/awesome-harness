    4  export class ReloadEventStore {
    5    private events: ReloadEvent[] = [];
    6    private seq = 0;
    7    private maxSize: number;
    8    private lastRecorded: Map<string, number> = new Map();
    9
   10    constructor(maxSize = 200) {
   11      this.maxSize = maxSize;
   12    }
   13
   14    record(workspaceId: string, reason: ReloadReason, trigger?: ReloadTrigger): ReloadEvent {
   15      const event: ReloadEvent = {
   16        id: shortId(),
   17        seq: ++this.seq,
   18        workspaceId,
   19        reason,
   20        trigger,
   21        timestamp: Date.now(),
   22      };
   23
   24      this.events.push(event);
   25      if (this.events.length > this.maxSize) {
   26        this.events.splice(0, this.events.length - this.maxSize);
   27      }
   28
   29      return event;
   30    }
   31
   32    recordDebounced(
   33      workspaceId: string,
   34      reason: ReloadReason,
   35      trigger?: ReloadTrigger,
   36      debounceMs = 750,
   37    ): ReloadEvent | null {
   38      const now = Date.now();
   39      const key = `${workspaceId}:${reason}`;
   40      const last = this.lastRecorded.get(key) ?? 0;
   41      if (now - last < debounceMs) return null;
   42      this.lastRecorded.set(key, now);
   43      return this.record(workspaceId, reason, trigger);
   44    }
   45
   46    list(workspaceId: string, since?: number): ReloadEvent[] {
   47      const cursor = Number.isFinite(since) ? (since as number) : 0;
   48      return this.events.filter((event) => event.workspaceId === workspaceId && event.seq > cursor);
   49    }
   50
   51    cursor(): number {
   52      return this.seq;
   53    }
