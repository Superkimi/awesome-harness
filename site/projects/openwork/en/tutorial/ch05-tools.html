   11  export function normalizePluginSpec(spec: string): string {
   12    const trimmed = spec.trim();
   13    if (trimmed.startsWith("file:") || trimmed.startsWith("http:") || trimmed.startsWith("https:") || trimmed.startsWith("git:")) {
   14      return trimmed;
   15    }
   16    if (trimmed.startsWith("/")) {
   17      return trimmed;
   18    }
   19    if (trimmed.startsWith("@")) {
   20      const atIndex = trimmed.indexOf("@", 1);
   21      return atIndex > 0 ? trimmed.slice(0, atIndex) : trimmed;
   22    }
   23    const atIndex = trimmed.indexOf("@");
   24    return atIndex > 0 ? trimmed.slice(0, atIndex) : trimmed;
   25  }

   34  async function listPluginFiles(dir: string, scope: "project" | "global", workspaceRoot?: string): Promise<PluginItem[]> {
   35    if (!(await exists(dir))) return [];
   36    const entries = await readdir(dir, { withFileTypes: true });
   37    const items: PluginItem[] = [];
   38    for (const entry of entries) {
   39      if (!entry.isFile()) continue;
   40      if (!entry.name.endsWith(".js") && !entry.name.endsWith(".ts")) continue;
   41      const absolutePath = join(dir, entry.name);
   42      const relativePath = workspaceRoot ? relative(workspaceRoot, absolutePath) : absolutePath;
   43      items.push({
   44        spec: `file://${absolutePath}`,
   45        source: scope === "project" ? "dir.project" : "dir.global",
   46        scope,
   47        path: relativePath,
   48      });
   49    }
   50    return items;
   51  }

   53  export async function listPlugins(serverConfig: ServerConfig, workspaceId: string, workspaceRoot: string, includeGlobal: boolean): Promise<{ items: PluginItem[]; loadOrder: string[] }> {
   54    const { data: config } = await readJsoncFile(opencodeConfigPath(workspaceRoot), {} as Record<string, unknown>, { allowInvalid: true });
   55    const pluginSpecs = pluginListFromConfig(config);
   56    const runtimeSpecs = runtimePluginList(await readRuntimeOpencodeConfig(serverConfig, workspaceId));
   57    const items: PluginItem[] = pluginSpecs.map((spec) => ({
   58      spec,
   59      source: "config",
   60      scope: "project",
   61    }));
   62
   63    const seen = new Set(pluginSpecs.map((spec) => normalizePluginSpec(spec)));
   64    for (const spec of runtimeSpecs) {
   65      const normalized = normalizePluginSpec(spec);
   66      if (seen.has(normalized)) continue;
   67      seen.add(normalized);
   68      items.push({
   69        spec,
   70        source: "config",
   71        scope: "project",
   72      });
   73    }
   74
   75    const projectDir = projectPluginsDir(workspaceRoot);
   76    items.push(...(await listPluginFiles(projectDir, "project", workspaceRoot)));
   77
   78    if (includeGlobal) {
   79      const globalDir = join(homedir(), ".config", "opencode", "plugins");
   80      items.push(...(await listPluginFiles(globalDir, "global")));
   81    }
   82
   83    return {
   84      items,
   85      loadOrder: ["config.global", "config.project", "dir.global", "dir.project"],
   86    };
   87  }

   89  export async function addPlugin(serverConfig: ServerConfig, workspaceId: string, spec: string): Promise<boolean> {
   90    validatePluginSpec(spec);
   91    const runtimeConfig = await readRuntimeOpencodeConfig(serverConfig, workspaceId);
   92    const pluginSpecs = runtimePluginList(runtimeConfig);
   93    const normalized = normalizePluginSpec(spec);
   94    const existing = pluginSpecs.find((item) => normalizePluginSpec(item) === normalized);
   95    if (existing) return false;
   96    pluginSpecs.push(spec);
   97    await writeRuntimeOpencodeConfig(serverConfig, workspaceId, (current) => ({ ...current, plugin: pluginSpecs }));
   98    return true;
