01 · Execution & sandbox

SOURCE FACT

MCP Apps run in sandboxed iframes with an allow-listed CSP. This protects the browser surface; it is not an OS sandbox around the OpenCode executor.

   10  function sourceList(values: string[], fallback: string): string {
   11    return values.length ? values.join(" ") : fallback;
   12  }
   13
   14  function safeOrigin(value: unknown): value is string {
   15    if (typeof value !== "string"
   16      || /\s/u.test(value)
   17      || value.includes(";")
   18      || value.includes("'")
   19      || value.includes(String.fromCharCode(34))) return false;
   20    try {
   21      const url = new URL(value);
   22      return (url.protocol === "https:" || url.protocol === "http:") && url.origin === value;
   23    } catch {
   24      return false;
   25    }
   26  }
   27
   28  export function parseMcpAppSandboxCsp(value: string | null): McpAppSandboxCsp {
   29    if (!value || value.length > MAX_CSP_QUERY_BYTES) {
   30      return { connectDomains: [], resourceDomains: [], frameDomains: [], baseUriDomains: [] };
   31    }
   32    try {
   33      const parsed = JSON.parse(value) as Partial<Record<keyof McpAppSandboxCsp, unknown>>;
   34      const domains = (key: keyof McpAppSandboxCsp) => Array.isArray(parsed[key])
   35        ? parsed[key].filter(safeOrigin).slice(0, 16)
   36        : [];
   37      return {
   38        connectDomains: domains("connectDomains"),
   39        resourceDomains: domains("resourceDomains"),
   40        frameDomains: domains("frameDomains"),
   41        baseUriDomains: domains("baseUriDomains"),
   42      };
   43    } catch {
   44      return { connectDomains: [], resourceDomains: [], frameDomains: [], baseUriDomains: [] };
   45    }
   46  }
   47
   48  export function buildMcpAppSandboxCsp(csp: McpAppSandboxCsp): string {
   49    const resources = csp.resourceDomains.join(" ");
   50    const withResources = (source: string) => resources ? `${source} ${resources}` : source;
   51    return [
   52      "default-src 'none'",
   53      `script-src ${withResources("'self' 'unsafe-inline'")}`,
   54      `style-src ${withResources("'self' 'unsafe-inline'")}`,
   55      `connect-src ${sourceList(csp.connectDomains, "'none'")}`,
   56      `img-src ${withResources("'self' data: blob:")}`,
   57      `font-src ${withResources("'self' data:")}`,
   58      `media-src ${withResources("'self' blob:")}`,
   59      `frame-src ${sourceList(["'self'", ...csp.frameDomains], "'self'")}`,
   60      `base-uri ${sourceList(csp.baseUriDomains, "'self'")}`,
   61      `worker-src ${withResources("'self' blob:")}`,
   62      "object-src 'none'",
   63      "form-action 'none'",
   64    ].join("; ");
   65  }

   68  (() => {
   69    if (window.self === window.top) throw new Error("Invalid MCP App sandbox embedding context.");
   70    const declaredHostOrigin = new URL(window.location.href).searchParams.get("hostOrigin");
   71    const referrerOrigin = document.referrer ? new URL(document.referrer).origin : null;
   72    if (declaredHostOrigin && referrerOrigin && declaredHostOrigin !== referrerOrigin) throw new Error("MCP App sandbox host origin mismatch.");
   73    const hostOrigin = referrerOrigin || declaredHostOrigin;
   74    if (!hostOrigin) throw new Error("MCP App sandbox host origin is unavailable.");
   75    const hostTargetOrigin = hostOrigin === "null" ? "*" : hostOrigin;
   76    const ownOrigin = window.location.origin;
   77    const inner = document.createElement("iframe");
   78    inner.title = "MCP App view";
   79    inner.style.cssText = "display:block;width:100%;height:100%;border:0;background:transparent";
   80    inner.setAttribute("sandbox", "allow-scripts allow-same-origin");
   81    document.body.appendChild(inner);
   82    window.addEventListener("message", (event) => {
   83      if (event.source === window.parent) {
   84        if (event.origin !== hostOrigin) return;
   85        if (event.data?.method === "ui/notifications/sandbox-resource-ready") {
   86          const html = event.data?.params?.html;
   87          const sandbox = event.data?.params?.sandbox;
   88          if (typeof sandbox === "string" && /^(?:allow-scripts|allow-same-origin|\s)+$/.test(sandbox)) inner.setAttribute("sandbox", sandbox);
   89          if (typeof html === "string") inner.srcdoc = html;
   90          return;
   91        }
   92        inner.contentWindow?.postMessage(event.data, "*");
   93        return;
   94      }
