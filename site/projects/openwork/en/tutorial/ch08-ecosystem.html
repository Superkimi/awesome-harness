  134      dirs.push({ dir: claudeDir, scope: "project" });

  136

  139      const globalClaude = join(homedir(), ".claude", "skills");
  140      const globalAgents = join(homedir(), ".agents", "skills");
  141      const globalAgentLegacy = join(homedir(), ".agent", "skills");
  142      dirs.push({ dir: globalOpenWork, scope: "global" });
  143      dirs.push({ dir: globalClaude, scope: "global" });
  144      dirs.push({ dir: globalAgents, scope: "global" });
  145      dirs.push({ dir: globalAgentLegacy, scope: "global" });
  146    }
  147
  148    const groups = await Promise.all(dirs.map(({ dir, scope }) => listSkillsInDir(dir, scope)));
  149    const items = groups.flat();
  150
  151    const seen = new Set<string>();
  152    return items.filter((item) => {
  153      if (seen.has(item.name)) return false;
  154      seen.add(item.name);

  156    });
