  297  def _is_data_url(url: str) -> bool:
  298      """Return whether `url` is an inline `data:` URL.
  299
  300      Any `data:` URL is treated as inline media to offload, because the XML
  301      history renderer drops `data:` URL blocks entirely (only `http(s)`-style
  302      references survive). This covers both base64 (`data:<mime>;base64,<payload>`)
  303      and percent-encoded / plaintext (`data:<mime>,<payload>`, e.g. an inline SVG)
  304      forms; whether the payload actually decodes is left to `_decode_data_url`.
  305      """
  306      return url.startswith("data:")
  307
  308
  309  def _extract_data_url(block: Any) -> str | None:  # noqa: ANN401
  310      """Return the embedded `data:` URL for an inline-media content block.
  311
  312      Detects the three inline-data content-block shapes that appear across
  313      LangChain messages:
  314
  315      1. A standard content block with an explicit `base64` field.
  316      2. A `data:` URL on the `url` field.
  317      3. An OpenAI-style `image_url` block whose `url` is a `data:` URL.
  318
  319      Both base64 (`;base64,`) and percent-encoded / plaintext `data:` URLs are
  320      detected -- e.g. an inline SVG (`data:image/svg+xml,<svg .../>`) -- because
  321      the XML history renderer drops *any* inline `data:` URL, so all of them must
  322      be offloaded to a referenceable path rather than left inline.
  323
  324      Shape 3 is defensive: `content_blocks` normalizes most `image_url` blocks
  325      (a base64 `data:` URL becomes shape 1; an `https` URL becomes a plain `url`
  326      image block), so this branch rarely fires for normalized input; it is kept
  327      for raw, un-normalized blocks.
  328
  329      This is pure detection and never raises: it reports *whether* a block
  330      carries inline data, leaving decoding (which can fail) to `_decode_data_url`.
  331
  332      Args:
  333          block: A single content block (usually a dict).
  334
  335      Returns:
  336          The block's `data:` URL, or `None` if the block carries no inline data.
  337      """
  338      if not isinstance(block, dict):
  339          return None
  340
  341      # 1. Standard content block with an explicit base64 field.
  342      raw_b64 = block.get("base64")
  343      if raw_b64:
  344          mime = block.get("mime_type") or "application/octet-stream"
  345          return f"data:{mime};base64,{raw_b64}"
  346
  347      # 2. Top-level data: URL.
  348      url = block.get("url", "")
  349      if isinstance(url, str) and _is_data_url(url):
  350          return url
  351
  352      # 3. OpenAI-style image_url with a data: URL.
  353      image_url = block.get("image_url")
  354      if isinstance(image_url, dict):
  355          inner = image_url.get("url", "")
  356          if isinstance(inner, str) and _is_data_url(inner):
  357              return inner
  358
  359      return None
