   25  def _normalize_args(args) -> list[str]:
   26      """Normalize args to list of strings."""
   27      if args is None:
   28          return []
   29      if isinstance(args, str):
   30          try:
   31              parsed = json.loads(args)
   32              return (
   33                  [str(x) for x in parsed]
   34                  if isinstance(parsed, list)
   35                  else [args]
   36              )
   37          except json.JSONDecodeError:
   38              return [x.strip() for x in args.split(",") if x.strip()]
   39      if isinstance(args, list):
   40          return [str(x) for x in args]
   41      return []

   44  def _normalize_mcp(mcp: dict) -> dict:
   45      """Normalize MCP server config."""
   46      out = dict(mcp)
   47      if "args" in out:
   48          out["args"] = _normalize_args(out["args"])

   56  def read_mcp_config() -> dict:
   57      """Read MCP config from ~/.eigent/mcp.json."""
   58      if not MCP_CONFIG_PATH.exists():
   59          default = {"mcpServers": {}}
   60          write_mcp_config(default)
   61          return default
   62      try:
   63          data = MCP_CONFIG_PATH.read_text(encoding="utf-8")
   64          parsed = json.loads(data)
   65          if not isinstance(parsed.get("mcpServers"), dict):
   66              return {"mcpServers": {}}
   67          for name, server in parsed["mcpServers"].items():
   68              if isinstance(server, dict):
   69                  parsed["mcpServers"][name] = _normalize_mcp(server)
   70          return parsed
   71      except (json.JSONDecodeError, OSError) as e:
   72          logger.warning("Failed to read MCP config: %s, using default", e)
   73          return {"mcpServers": {}}

   76  def write_mcp_config(config: dict) -> None:
   77      """Write MCP config to ~/.eigent/mcp.json."""
   78      MCP_CONFIG_DIR.mkdir(parents=True, exist_ok=True)
   79      MCP_CONFIG_PATH.write_text(
   80          json.dumps(config, indent=2, ensure_ascii=False),
   81          encoding="utf-8",
   82      )
