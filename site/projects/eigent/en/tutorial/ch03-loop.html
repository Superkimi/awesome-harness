   24  # This ensures each request has its own event loop reference, avoiding race conditions
   25  _main_event_loop_var: contextvars.ContextVar[
   26      asyncio.AbstractEventLoop | None
   27  ] = contextvars.ContextVar("_main_event_loop", default=None)
   28
   29  # Global fallback for main event loop reference
   30  # Used when contextvars don't propagate to worker threads (e.g., asyncio.to_thread)
   31  _GLOBAL_MAIN_LOOP: asyncio.AbstractEventLoop | None = None
   32  _GLOBAL_MAIN_LOOP_LOCK = Lock()
   33
   34
   35  def set_main_event_loop(loop: asyncio.AbstractEventLoop | None):
   36      """Set the main event loop reference for thread-safe task scheduling.
   37
   38      This should be called from the main async context before spawning threads
   39      that need to schedule async tasks. Uses both contextvars (for request isolation)
   40      and a global fallback (for thread pool workers where contextvars may not propagate).
   41      """
   42      global _GLOBAL_MAIN_LOOP
   43      _main_event_loop_var.set(loop)
   44      with _GLOBAL_MAIN_LOOP_LOCK:
   45          _GLOBAL_MAIN_LOOP = loop
   46
   47
   48  def _get_registered_main_loop() -> asyncio.AbstractEventLoop | None:
   49      main_loop = _main_event_loop_var.get()
   50      if main_loop is None:
   51          with _GLOBAL_MAIN_LOOP_LOCK:
   52              main_loop = _GLOBAL_MAIN_LOOP
   53      if main_loop is not None and main_loop.is_running():
   54          return main_loop
   55      return None
