   58  def _build_single_agent_context(
   59      task_lock: TaskLock,
   60      project_context: str | None = None,
   61      current_user_prompt: str = "",
   62  ) -> str:
   63      # 1. Durable cross-restart context from LocalMemoryStore (M4 path).
   64      durable = build_durable_context_for_task_lock(
   65          task_lock,
   66          mode="single_agent",
   67          current_user_prompt=current_user_prompt,
   68          token_budget=_MEMORY_TOKEN_BUDGET,
   69      )
   70      if durable:
   71          return durable + "\n\n"
   72
   73      # 2. In-process conversation history (hot follow-up turns).
   74      if getattr(task_lock, "conversation_history", None):
   75          lines = ["=== Previous Conversation ==="]
   76          for entry in task_lock.conversation_history:
   77              role = entry.get("role", "")
   78              content = entry.get("content", "")
   79              if role == "task_result" and isinstance(content, dict):
   80                  task_content = content.get("task_content")
   81                  task_result = content.get("task_result")
   82                  if task_content:
   83                      lines.append(f"Previous task: {task_content}")
   84                  if task_result:
   85                      lines.append(f"Previous result: {task_result}")
   86              elif content:
   87                  lines.append(f"{role}: {content}")
   88          memory_context = build_memory_context(task_lock)
   89          if memory_context:
   90              lines.append(memory_context.rstrip())
   91          lines.append("=== End Previous Conversation ===")
   92          return "\n".join(lines) + "\n\n"
   93
   94      # 3. Phase-0 bridge fallback (frontend-sent project_context).
   95      durable_context = (project_context or "").strip()
   96      if not durable_context:
   97          return ""
   98      return (
   99          "=== Persisted Project Context ===\n"
  100          f"{durable_context}\n"
  101          "=== End Persisted Project Context ===\n\n"
  102      )
