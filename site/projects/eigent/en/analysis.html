   48    frontmatter: string | null;
   49    body: string;
   50  } {
   51    // Strip BOM and leading whitespace/newlines so the first `---` is detected
   52    const cleaned = contents.replace(/^\uFEFF/, '').trimStart();
   53    const lines = cleaned.split('\n');
   54    if (!lines.length || lines[0].trim() !== FRONTMATTER_DELIM) {
   55      return { frontmatter: null, body: cleaned };
   56    }
   57    for (let i = 1; i < lines.length; i++) {
   58      if (lines[i].trim() === FRONTMATTER_DELIM) {
   59        const frontmatter = lines.slice(1, i).join('\n');
   60        const body = lines.slice(i + 1).join('\n');
   61        return { frontmatter, body };
   62      }
   63    }
   64    return { frontmatter: null, body: cleaned };

   69    const out: Record<string, string> = {};

   78          .replace(/\\"/g, '"')

   80      }
