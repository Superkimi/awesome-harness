   35  def set_main_event_loop(loop: asyncio.AbstractEventLoop | None):
   36      """Set the main event loop reference for thread-safe task scheduling.
   37
   38      This should be called from the main async context before spawning threads
   39      that need to schedule async tasks. Uses both contextvars (for request isolation)
   40      and a global fallback (for thread pool workers where contextvars may not propagate).
   41      """
   42      global _GLOBAL_MAIN_LOOP
   43      _main_event_loop_var.set(loop)
   44      with _GLOBAL_MAIN_LOOP_LOCK:
   45          _GLOBAL_MAIN_LOOP = loop
