先用一个生活比喻

指令像机场值机单：先认出航班，再检查登机规则，交给正确柜台执行，最后把登机和到达状态写回系统。

先回答本章问题

一个内置指令如何被解析、路由、执行并写回会话？ 把 Eigent 想成一个带项目记忆的办公室：任务锁是工单，Agent 是员工，MCP 是工具柜，记忆服务负责交接班。

不要把一段代码当成整个系统；下面每张卡分别说明事实、白话、工程影响、边界和可点击的固定源码。

   43  /**
   44   * Split YAML frontmatter from the body of a SKILL.md file.
   45   * Expects content to start with "---", then YAML, then "---", then body.
   46   */
   47  export function splitFrontmatter(contents: string): {
   48    frontmatter: string | null;
   49    body: string;
   50  } {
   51    // Strip BOM and leading whitespace/newlines so the first `---` is detected
   52    const cleaned = contents.replace(/^\uFEFF/, '').trimStart();
   53    const lines = cleaned.split('\n');
   54    if (!lines.length || lines[0].trim() !== FRONTMATTER_DELIM) {
   55      return { frontmatter: null, body: cleaned };
   56    }
   57    for (let i = 1; i < lines.length; i++) {
   58      if (lines[i].trim() === FRONTMATTER_DELIM) {
   59        const frontmatter = lines.slice(1, i).join('\n');
   60        const body = lines.slice(i + 1).join('\n');
   61        return { frontmatter, body };
   62      }
   63    }
   64    return { frontmatter: null, body: cleaned };
   65  }

   67  /** Simple YAML-like parse for "name:" and "description:" (first-level keys only). */
   68  function parseSimpleYaml(text: string): Record<string, string> {
   69    const out: Record<string, string> = {};
   70    const lines = text.split('\n');
   71    for (const line of lines) {
   72      const match = line.match(/^\s*([a-zA-Z_][a-zA-Z0-9_]*)\s*:\s*(.*)$/);
   73      if (match) {
   74        const value = match[2].trim();
   75        // Lowercase key so `Name:` / `name:` / `NAME:` all work
   76        out[match[1].toLowerCase()] = value
   77          .replace(/^['"]|['"]$/g, '')
   78          .replace(/\\"/g, '"')
   79          .trim();
   80      }
   81    }
   82    return out;
   83  }
