   40      this.cancelRequested.delete(workflowId)

   42      for (const [token, pending] of this.pendingApprovals) {
   43        if (pending.entry.runId === runId) this.pendingApprovals.delete(token)
   44      }

   46        this.cleanupTimers.delete(timer)
   47        this.liveNodeStatus.delete(workflowId)
   48        this.liveNodeResults.delete(workflowId)
   49      }, lingerMs)

   51      this.cleanupTimers.add(timer)
   52    }

   55      const live = new Map<string, WorkflowNodeRunStatus>([[nodeId, 'running']])
   56      this.liveNodeStatus.set(workflowId, live)

   77    }
