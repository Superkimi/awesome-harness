01 · Session 用完整 snapshot 加 append-only JSONL journal

SOURCE FACT

save 比较 metadata/vector dirty state：需要全量时 checkpoint snapshot 并删 journal，否则只追加 messages/env snapshots/memory injections/replay events delta；journal 超上限或 append 失败再回退 snapshot。

PLAIN LANGUAGE

平时只往流水账追加新变化，偶尔把整本账重抄成快照；这样频繁保存不会每次重写全部历史。

IMPLICATION

兼顾耐久与 I/O；多写者必须依靠更上层会话所有权避免交错 append。

02 · 损坏 journal 跳过坏行、挽救 glued entries，并在下次保存自愈

SOURCE FACT

replay 不在首个 parse error 停止，而是继续后续行；对 torn append 与下一条粘在同一行的情况扫描结构起点并 stream-parse。发现腐坏后备份 corrupt.jsonl，强制下次 checkpoint 把已挽救状态固化。

IMPLICATION

长会话恢复质量高，且保留法证副本。

   26  /// Attempt to recover complete entries from a journal line that failed the
   27  /// strict one-entry-per-line parse.
   28  ///
   29  /// If a writer died mid-append (torn line without a trailing newline), the
   30  /// next successful append starts writing on the same line, producing
   31  /// `<torn json><complete entry json>\n` or `<entry json><entry json>\n`.
   32  /// Serialized entries always begin with `{"meta":` (struct field order), so
   33  /// scan for candidate starts and stream-parse consecutive complete entries
   34  /// from the first position that yields any.
   35  fn salvage_glued_journal_entries(line: &str, mut apply: impl FnMut(SessionJournalEntry)) -> usize {
   36      const ENTRY_START: &str = "{\"meta\":";
   37      let mut salvaged = 0usize;
   38      let mut search_from = 0usize;
   39      while let Some(rel) = line
   40          .get(search_from..)
   41          .and_then(|rest| rest.find(ENTRY_START))
   42      {
   43          let candidate_start = search_from + rel;
   44          let mut stream = serde_json::Deserializer::from_str(&line[candidate_start..])
   45              .into_iter::<SessionJournalEntry>();
   46          let mut parsed = Vec::new();
   47          for item in &mut stream {
   48              match item {
   49                  Ok(entry) => parsed.push(entry),
   50                  Err(_) => break,
   51              }
   52          }
   53          if !parsed.is_empty() {
   54              salvaged += parsed.len();
   55              for entry in parsed {
   56                  apply(entry);
   57              }
   58              break;
   59          }
   60          search_from = candidate_start + ENTRY_START.len();
   61      }
   62      salvaged
   63  }

   65  /// Replay every parseable entry from a session journal, tolerating corrupt
   66  /// lines instead of truncating the transcript at the first bad byte.
   67  ///
   68  /// Journals are append-only JSONL written by `append_json_line_fast`. A crash,
   69  /// full disk, or (historically) interleaved multi-write appends can leave a
   70  /// torn or glued line behind. The old replay loop stopped at the first parse
   71  /// failure, silently dropping every later entry, which surfaced as "my last
   72  /// prompt is missing" after resuming a long session. Skipping only the bad
   73  /// line preserves the rest of the transcript; per-entry `meta` snapshots make
   74  /// later entries self-sufficient for metadata, and appended messages after a
   75  /// gap are far better than losing the whole tail.
   76  fn replay_journal_lines(
   77      journal_path: &Path,
   78      mut apply: impl FnMut(SessionJournalEntry),
   79  ) -> Result<JournalReplayStats> {
   80      let mut stats = JournalReplayStats::default();
   81      if !journal_path.exists() {
   82          return Ok(stats);
   83      }
   84
   85      let file = std::fs::File::open(journal_path)?;
   86      let reader = BufReader::new(file);
   87      for (line_idx, line) in reader.lines().enumerate() {
   88          let line = line?;
   89          let trimmed = line.trim();
   90          if trimmed.is_empty() {
   91              continue;
   92          }
   93          match serde_json::from_str::<SessionJournalEntry>(trimmed) {
   94              Ok(entry) => {
   95                  stats.entries += 1;
   96                  apply(entry);
   97              }
   98              Err(err) => {
   99                  stats.skipped_lines += 1;
  100                  let salvaged = salvage_glued_journal_entries(trimmed, &mut apply);
  101                  stats.entries += salvaged;
  102                  stats.salvaged_entries += salvaged;
  103                  crate::logging::warn(&format!(
  104                      "Session journal parse failed at {} line {} ({}); salvaged {} glued entr{} and continuing replay",
  105                      journal_path.display(),
  106                      line_idx + 1,
  107                      err,
  108                      salvaged,
  109                      if salvaged == 1 { "y" } else { "ies" }
  110                  ));
  111              }
  112          }
  113      }
  114
  115      if stats.is_corrupt() {
  116          crate::logging::event_warn(
  117              "SESSION_PERSISTENCE",
  118              vec![
  119                  ("phase", "journal_replay_corruption".to_string()),
  120                  ("path", journal_path.display().to_string()),
  121                  ("entries_replayed", stats.entries.to_string()),
  122                  ("lines_skipped", stats.skipped_lines.to_string()),
  123                  ("entries_salvaged", stats.salvaged_entries.to_string()),
  124              ],
  125          );
  126      }
  127
  128      Ok(stats)
  129  }
