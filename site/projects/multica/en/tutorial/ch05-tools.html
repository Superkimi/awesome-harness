   11  //
   12  // Spec: https://docs.composio.dev/reference/api-reference/tools/postToolsExecuteByToolSlug
   13  //
   14  // Either ConnectedAccountID or (UserID + the tool's toolkit) is required so
   15  // Composio knows which credential set to use. The SDK does not enforce that
   16  // invariant up front; the upstream returns a 422 with a clear message when
   17  // missing.
   18  type ExecuteToolRequest struct {
   19  	// Arguments is the structured input to the tool. Shape varies per tool.
   20  	Arguments map[string]any `json:"arguments,omitempty"`
   21
   22  	// ConnectedAccountID pins execution to a specific connected account.
   23  	ConnectedAccountID string `json:"connected_account_id,omitempty"`
   24
   25  	// UserID lets Composio resolve the connected account by user when
   26  	// the caller does not have the explicit `ca_` id handy.
   27  	UserID string `json:"user_id,omitempty"`
   28
   29  	// Version pins the tool definition version. Pass "latest" or a dated
   30  	// version like "20251027_00"; defaults to "00000000_00" upstream.
   31  	//
   32  	// The Composio docs note that manual tool execution requires an explicit
   33  	// version; setting this avoids unintended drift when Composio promotes
   34  	// a new latest.
   35  	Version string `json:"version,omitempty"`

   39  	// Deprecated: marked deprecated on the Composio side (v3.1) — kept here

   56  // ExecuteTool calls a Composio tool by its slug
   57  // (SCREAMING_SNAKE_CASE, e.g. GITHUB_CREATE_ISSUE).
   58  //
   59  // This is the deterministic backend path — it skips MCP/session orchestration
   60  // and is the right call for fixed flows like autopilots or built-in skills.
   61  func (c *Client) ExecuteTool(ctx context.Context, toolSlug string, req ExecuteToolRequest) (*ExecuteToolResponse, error) {
   62  	if toolSlug == "" {
   63  		return nil, errors.New("composio: ExecuteTool: toolSlug is required")
   64  	}
   65  	if req.ConnectedAccountID == "" && req.UserID == "" {
   66  		return nil, errors.New("composio: ExecuteTool: either ConnectedAccountID or UserID must be set")
   67  	}
   68  	var out ExecuteToolResponse
   69  	if err := c.do(c.newRequest(ctx).SetBody(req),
   70  		http.MethodPost, "/tools/execute/"+url.PathEscape(toolSlug), &out); err != nil {
   71  		return nil, err
   72  	}
   73  	return &out, nil
   74  }
