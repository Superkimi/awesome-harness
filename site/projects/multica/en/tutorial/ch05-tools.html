   56  // ExecuteTool calls a Composio tool by its slug
   57  // (SCREAMING_SNAKE_CASE, e.g. GITHUB_CREATE_ISSUE).
   58  //
   59  // This is the deterministic backend path — it skips MCP/session orchestration
   60  // and is the right call for fixed flows like autopilots or built-in skills.
   61  func (c *Client) ExecuteTool(ctx context.Context, toolSlug string, req ExecuteToolRequest) (*ExecuteToolResponse, error) {
   62  	if toolSlug == "" {
   63  		return nil, errors.New("composio: ExecuteTool: toolSlug is required")
   64  	}
   65  	if req.ConnectedAccountID == "" && req.UserID == "" {
   66  		return nil, errors.New("composio: ExecuteTool: either ConnectedAccountID or UserID must be set")
   67  	}
   68  	var out ExecuteToolResponse
   69  	if err := c.do(c.newRequest(ctx).SetBody(req),
   70  		http.MethodPost, "/tools/execute/"+url.PathEscape(toolSlug), &out); err != nil {
   71  		return nil, err
   72  	}
   73  	return &out, nil
   74  }
