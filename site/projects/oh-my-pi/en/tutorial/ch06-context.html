  176  /**
  177   * Collect superseded tool results: for every unpruned, unprotected tool result
  178   * whose paired call resolves a supersede key, a LATER result with the same key
  179   * — or with a key that is the `"\u0000"`-prefix parent of this one — marks it
  180   * superseded. Returned in message order.
  181   */
  182  function collectSupersededResults(
  183  	entries: readonly SessionEntry[],
  184  	toolCallsById: ReadonlyMap<string, AgentToolCall>,
  185  	supersedeKey: SupersedeKeyFn,
  186  	protectedTools: readonly ProtectedToolMatcher[],
  187  ): SupersedeCandidate[] {
  188  	const candidates: SupersedeCandidate[] = [];
  189  	const seenKeys = new Set<string>();
  190  	for (let i = entries.length - 1; i >= 0; i--) {
  191  		const entry = entries[i];
  192  		const message = getToolResultMessage(entry);
  193  		if (!message || message.prunedAt !== undefined) continue;
  194  		const toolCall = toolCallsById.get(message.toolCallId);
  195  		if (!toolCall) continue;
  196  		if (isProtectedToolResult(message, toolCall, protectedTools)) continue;
  197  		const key = supersedeKey(toolCall.name, toolCall.arguments as Record<string, unknown>);
  198  		if (key === undefined) continue;
  199  		const separator = key.indexOf("\u0000");
  200  		const superseded = seenKeys.has(key) || (separator >= 0 && seenKeys.has(key.slice(0, separator)));
  201  		seenKeys.add(key);
  202  		if (!superseded) continue;
  203  		candidates.push({
  204  			entry: entry as SessionMessageEntry,
  205  			message,
  206  			index: i,
  207  			tokens: estimateTokens(message as AgentMessage),
  208  			notice: SUPERSEDED_NOTICE,
  209  		});
  210  	}
  211  	return candidates.reverse();
  212  }

  215   * Collect tool results their tool flagged contextually useless (zero matches,
  216   * elapsed wait): unpruned, non-error, unprotected, not in `exclude`, and large
  217   * enough that blanking to {@link USELESS_NOTICE} actually saves tokens.
  218   * Returned in message order.
  219   */
  220  function collectUselessResults(
  221  	entries: readonly SessionEntry[],
  222  	toolCallsById: ReadonlyMap<string, AgentToolCall>,
  223  	protectedTools: readonly ProtectedToolMatcher[],
  224  	exclude: ReadonlySet<ToolResultMessage>,
  225  ): SupersedeCandidate[] {
  226  	const candidates: SupersedeCandidate[] = [];
  227  	for (let i = 0; i < entries.length; i++) {
  228  		const entry = entries[i];
  229  		const message = getToolResultMessage(entry);
  230  		if (message?.useless !== true || message.prunedAt !== undefined || message.isError === true) continue;
  231  		if (exclude.has(message)) continue;
  232  		if (isProtectedToolResult(message, toolCallsById.get(message.toolCallId), protectedTools)) continue;
  233  		const tokens = estimateTokens(message as AgentMessage);
  234  		if (estimatePrunedSavings(tokens, USELESS_NOTICE) <= 0) continue;
  235  		candidates.push({ entry: entry as SessionMessageEntry, message, index: i, tokens, notice: USELESS_NOTICE });
  236  	}
  237  	return candidates;
  238  }

  240  /**
  241   * Prune superseded tool results (e.g. stale `read` outputs replaced by a newer
  242   * read of the same file) and, when `pruneUseless` is set, results their tool
  243   * flagged contextually useless. Cheap, incremental, and prompt-cache-aware: a
  244   * candidate is pruned now only when the suffix after it is small (tail case —
  245   * the read→edit→read loop) or when the context has been idle long enough that
  246   * the provider cache is cold anyway (then all still-sent candidates flush).
  247   * Never mutates entries before `keepBoundaryId` (summarized away — not sent).
  248   */
  249  export function pruneSupersededToolResults(entries: SessionEntry[], config: SupersedePruneConfig): PruneResult {
  250  	const toolCallsById = collectToolCallsById(entries);
  251  	const candidates = config.supersedeKey
  252  		? collectSupersededResults(entries, toolCallsById, config.supersedeKey, config.protectedTools)
  253  		: [];
  254  	if (config.pruneUseless) {
  255  		const exclude = new Set(candidates.map(candidate => candidate.message));
  256  		candidates.push(...collectUselessResults(entries, toolCallsById, config.protectedTools, exclude));
  257  		candidates.sort((a, b) => a.index - b.index);
  258  	}
  259  	if (candidates.length === 0) return { prunedCount: 0, tokensSaved: 0 };
  260
  261  	const now = config.now ?? Date.now();
  262  	let lastMessageTimestamp: number | undefined;
  263  	for (let i = entries.length - 1; i >= 0; i--) {
  264  		const entry = entries[i];
  265  		if (entry.type !== "message") continue;
  266  		const timestamp = (entry.message as AgentMessage).timestamp;
  267  		if (typeof timestamp === "number") lastMessageTimestamp = timestamp;
  268  		break;
  269  	}
  270  	const idle =
  271  		lastMessageTimestamp !== undefined && now - lastMessageTimestamp >= (config.idleFlushMs ?? DEFAULT_IDLE_FLUSH_MS);
  272
  273  	const boundaryIndex = resolveBoundaryIndex(entries, config.keepBoundaryId);
  274
  275  	let toPrune: SupersedeCandidate[];
  276  	if (idle) {
  277  		// Provider cache is cold (idle exceeds the retention TTL), so re-writing
  278  		// the sent region costs nothing. Entries before the compaction boundary
  279  		// are summarized away and never sent — skip them to avoid pointless churn.
  280  		toPrune = candidates.filter(candidate => candidate.index >= boundaryIndex);
  281  	} else {
  282  		const suffixTokenLimit = config.suffixTokenLimit ?? DEFAULT_SUFFIX_TOKEN_LIMIT;
  283  		// suffixTokens[i] = estimated tokens of all messages strictly after entry i.
  284  		// Mutating a candidate re-writes its suffix in the warm cache, so prune only
  285  		// when that suffix is small (cheap-to-recache tail) and the candidate sits
  286  		// at/after the compaction boundary.
  287  		const suffixTokens = computeMessageSuffixTokens(entries);
  288  		toPrune = candidates.filter(
  289  			candidate => candidate.index >= boundaryIndex && suffixTokens[candidate.index] <= suffixTokenLimit,
  290  		);
  291  	}
  292  	if (toPrune.length === 0) return { prunedCount: 0, tokensSaved: 0 };
  293
  294  	const prunedAt = Date.now();
  295  	let tokensSaved = 0;
  296  	for (const candidate of toPrune) {
  297  		candidate.message.content = [{ type: "text", text: candidate.notice }];
  298  		candidate.message.prunedAt = prunedAt;
  299  		invalidateMessageCache(candidate.message as AgentMessage);
  300  		tokensSaved += estimatePrunedSavings(candidate.tokens, candidate.notice);
  301  	}
  302  	return { prunedCount: toPrune.length, tokensSaved };
  303  }
