   66        throw new Error('Cannot call `startExtension` prior to calling `start`.');

  463  async function findUpwardGeminiFiles(
  464    startDir: string,
  465    stopDir: string,
  466  ): Promise<string[]> {
  467    const upwardPaths: string[] = [];
  468    let currentDir = toAbsolutePath(startDir);
  469    const resolvedStopDirKey = normalizePath(stopDir);
  470    const geminiMdFilenames = getAllGeminiMdFilenames();
  471    const globalGeminiDirKey = normalizePath(path.join(homedir(), GEMINI_DIR));
  472
  473    debugLogger.debug(
  474      '[DEBUG] [MemoryDiscovery] Starting upward search from',
  475      currentDir,
  476      'stopping at',
  477      stopDir,
  478    );
  479
  480    while (true) {
  481      if (normalizePath(currentDir) === globalGeminiDirKey) {
  482        break;
  483      }
  484
  485      // Parallelize checks for all filename variants in the current directory
  486      const accessChecks = geminiMdFilenames.map(async (filename) => {
  487        const potentialPath = toAbsolutePath(path.join(currentDir, filename));
  488        try {
  489          await fs.access(potentialPath, fsSync.constants.R_OK);
