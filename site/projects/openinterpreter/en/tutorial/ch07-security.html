   97  pub(crate) fn bounded_walk(root: &Path) -> io::Result<Vec<WalkEntry>> {
   98      let root_metadata = fs::symlink_metadata(root)?;
   99      if root_metadata.file_type().is_symlink() {
  100          return Ok(Vec::new());
  101      }
  102      if root_metadata.is_file() {
  103          return Ok(vec![WalkEntry {
  104              path: root.to_path_buf(),
  105              kind: WalkEntryKind::File,
  106          }]);
  107      }
  108      if !root_metadata.is_dir() || is_ignored_dir(root) {
  109          return Ok(Vec::new());
  110      }
  111
  112      let mut entries = Vec::new();
  113      let mut queue = VecDeque::from([(root.to_path_buf(), 0usize)]);
  114      while let Some((dir, depth)) = queue.pop_front() {
  115          if depth >= MAX_WALK_DEPTH || entries.len() >= MAX_WALK_ENTRIES {
  116              continue;
  117          }
  118          for entry in fs::read_dir(&dir)? {
  119              let entry = entry?;
  120              let path = entry.path();
  121              let metadata = fs::symlink_metadata(&path)?;
  122              let file_type = metadata.file_type();
  123              if file_type.is_symlink() {
  124                  continue;
  125              }
  126              if file_type.is_file() {
  127                  entries.push(WalkEntry {
  128                      path,
  129                      kind: WalkEntryKind::File,
  130                  });
  131              } else if file_type.is_dir() && !is_ignored_dir(&path) {
  132                  entries.push(WalkEntry {
  133                      path: path.clone(),
  134                      kind: WalkEntryKind::Directory,
  135                  });
  136                  queue.push_back((path, depth + 1));
  137              }
  138              if entries.len() >= MAX_WALK_ENTRIES {
  139                  break;
  140              }
  141          }
  142      }
  143      Ok(entries)
  144  }

02 · 审批和能力授权是两条轴，OS 沙箱是真实进程变换

SOURCE FACT

IMPLICATION
