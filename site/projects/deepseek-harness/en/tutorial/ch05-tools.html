01 · Tool calls are grouped by executionMode before dispatch

SOURCE FACT

executeToolCalls parses every tool call, puts consecutive parallel calls in a bounded pool, and treats exclusive calls as barriers.

PLAIN LANGUAGE

Safe parallel reads can run together while conflict-prone writes form a queue.

IMPLICATION

Concurrency policy should be part of the tool declaration, not an ad hoc model decision.

02 · Tool results may finish asynchronously but commit in model order

SOURCE FACT

runGroup stores results in slots; commitReady finalizes contiguous indexes in order, appends tool results, and queues additionalContexts for the next step.

PLAIN LANGUAGE

A third tool finishing first cannot place its receipt before the second tool in model history.

IMPLICATION

Concurrent tools need stable call ids, commit order, and synthetic results for cancellation.

  117   * their contexts into the owning batch, records results for skipped calls, and
  118   * returns an aborted outcome. Scheduler failure drains dispatches without
  119   * committing synthetic recovery results.

  122    ctx: Context,
  123    turn: number,
  124    step: number,
  125    group: PlannedCall[],
  126    mode: ToolExecutionMode['kind'],
  127    signal: AbortSignal,
  128    acceptContext: (context: UserMessage) => void,
  129  ): Promise<GroupOutcome> {
  130    const { session } = ctx.agents.requireInitiator()
  131    const { maxParallelToolCalls } = ctx.agentLoop.config
  132    const slots: (Slot | undefined)[] = group.map(() => undefined)
  133    // Started slots retain their `tool/call` seq so the result can cite it.
  134    const callSeqs: number[] = group.map(() => -1)
  135    let nextToStart = 0
  136    let committed = 0
  137    let started = 0
  138    let aborted: boolean = signal.aborted
  139    let concluded = false
  140    let schedulerFailure: { error: unknown } | undefined
  141    const throwSchedulerFailure = (): void => {
  142      if (schedulerFailure !== undefined) throw schedulerFailure.error
  143    }
  144
  145    // `committed` advances only across contiguous model-order slots.
  146    const commitReady = async (): Promise<void> => {
  147      while (committed < group.length) {
  148        const slot = slots[committed]
  149        if (slot === undefined) break
  150        const call = group[committed]
  151        const result = slot.needsPost
  152          ? await ctx.tools[TOOL_RUNTIME_SCHEDULER].finalize(slot.exec, slot.result)
  153          : ctx.tools[TOOL_RUNTIME_SCHEDULER].finish(slot.exec, slot.result)
  154        // oxlint-disable-next-line typescript/no-non-null-assertion -- bounded index
  155        appendToolResult(session, turn, step, call!.block, result, callSeqs[committed]!)
  156        for (const context of result.additionalContexts ?? []) acceptContext(context)
  157        concluded ||= result.concludesTurn === true
  158        committed++
  159      }
  160    }

  162    const inFlight = new Map<number, Promise<number>>()
  163
  164    const startCall = async (index: number): Promise<void> => {
  165      // oxlint-disable-next-line typescript/no-non-null-assertion -- bounded index
  166      const call = group[index]!
  167      callSeqs[index] = appendToolCall(session, turn, step, call.block)
  168      started++
  169      const prepared = await ctx.tools[TOOL_RUNTIME_SCHEDULER].prepare(call.exec)
  170      throwSchedulerFailure()
  171      switch (prepared.kind) {
  172        case 'dispatch': {
  173          const promise = ctx.tools[TOOL_RUNTIME_SCHEDULER].dispatch(prepared.exec).then(
  174            (outcome) => {
  175              slots[index] = { exec: prepared.exec, result: outcome.result, needsPost: outcome.kind === 'post-result' }
  176              return index
  177            },
  178            (error: unknown) => {
  179              schedulerFailure ??= { error }
  180              return index
  181            },
  182          )
  183          inFlight.set(index, promise)
  184          break
  185        }
  186        case 'post-result':
  187          slots[index] = { exec: prepared.exec, result: prepared.result, needsPost: true }
  188          break
  189        case 'final-result':
  190          slots[index] = { exec: prepared.exec, result: prepared.result, needsPost: false }
  191          break
  192        /* v8 ignore next -- closed-union exhaustiveness guard */
  193        default:
  194          assertNever(prepared, 'tool-call scheduler prepare result')
  195      }
  196    }
  197
  198    const fillPool = async (): Promise<void> => {
  199      while (!aborted && nextToStart < group.length && inFlight.size < maxParallelToolCalls) {

  201        // oxlint-disable-next-line typescript/no-non-null-assertion -- bounded by the loop condition
  202        const nextCall = group[nextToStart]!

  204          && ctx.tools.executionMode(nextCall.exec).kind !== 'parallel') break

  206        nextToStart++

  208        await commitReady()
  209        throwSchedulerFailure()
  210        // Abort may arrive while pre-execute awaits.
  211        if (signal.aborted) aborted = true
  212      }
  213    }
  214
  215    // Ordered pre-execute may await; only dispatch/body overlaps. A scheduler
  216    // failure stops new dispatches and reaches the turn boundary after every
  217    // already-started dispatch settles.
  218    try {
  219      await fillPool()
  220      while (inFlight.size > 0) {
  221        const settledIndex = await Promise.race(inFlight.values())
  222        inFlight.delete(settledIndex)

  224        await commitReady()
  225        throwSchedulerFailure()
  226        // Abort may arrive while a tool or ordered commit awaits.
  227
  228        if (signal.aborted) aborted = true
  229        await fillPool()
  230      }
