   30  /// cells are on it equals the legacy `AGENT_FILENAMES` list exactly.
   31  fn find_agent_files(dir: &Path, filenames: &[&str]) -> Vec<PathBuf> {
   32      filenames
   33          .iter()
   34          .filter_map(|name| {

   36              path.exists().then_some(path)
   37          })
   38          .collect()

   41  /// Find `*.md` files in `.grok/rules/`, `.claude/rules/`, and `.cursor/rules/`,
   42  /// sorted alphabetically. `rules_subdirs` is the (compat-gated) list, precomputed
   43  /// once by the caller so the walk doesn't re-allocate it per directory.
   44  fn find_rules_files(dir: &Path, rules_subdirs: &[&str]) -> Vec<PathBuf> {
   45      let mut results = Vec::new();
   46      for rules_subdir in rules_subdirs {
   47          let rules_dir = dir.join(rules_subdir);
   48          if !rules_dir.is_dir() {
   49              continue;
   50          }
   51          let mut entries: Vec<PathBuf> = match std::fs::read_dir(&rules_dir) {
   52              Ok(iter) => iter
   53                  .filter_map(|entry| entry.ok())
   54                  .map(|e| e.path())
   55                  .filter(|p| {
   56                      p.extension()
   57                          .and_then(|ext| ext.to_str())
   58                          .is_some_and(|ext| ext.eq_ignore_ascii_case("md"))
   59                  })
   60                  .collect(),
   61              Err(_) => continue,
   62          };
   63          entries.sort_by(|a, b| a.file_name().cmp(&b.file_name()));
   64          results.extend(entries);
   65      }
   66      results
   67  }

02 · PromptContext 是可序列化、可检查的一等契约

SOURCE FACT

PromptContext 保存版本、模式、受众、模板、指令文件、memory、角色、OS/shell/cwd/date 与交互模式；Primary/Subagent 选择不同基础模板，但子 Agent 仍拿到完整 AGENTS.md。

PLAIN LANGUAGE

系统提示词的输入不是散落变量，而是一张可以导出检查的配置表。

IMPLICATION

Prompt provenance 和可重放性明显优于只在运行时拼字符串。
